**python/src/gemini_sdk/process/spawn_per_call.py:**
```python
import subprocess
import sys
from typing import Any
import anyio
from anyio.abc import Process

class SpawnPerCallStrategy:
    """Default ProcessStrategy: spawns one gemini-cli process per query call."""

    async def spawn(
        self, argv: list[str], env: dict[str, str], **kwargs: Any
    ) -> Process:
        creation_flags = 0
        if sys.platform == "win32":
            # FDN-05: Hide console window on Windows
            creation_flags = subprocess.CREATE_NO_WINDOW
            # FDN-03: Windows .cmd shims need shell execution
            # Unlike TS, Python's subprocess handles .cmd via shell=True
            # But anyio.open_process does not have a shell= parameter
            # Instead, on Windows we pass the command through cmd.exe
            cmd = " ".join(f'"{a}"' for a in argv)
            return await anyio.open_process(
                cmd,
                env=env,
                stdin=subprocess.PIPE,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                shell=True,
                creationflags=creation_flags,
                **kwargs,
            )
        return await anyio.open_process(
            argv,
            env=env,
            stdin=subprocess.PIPE,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            **kwargs,
        )
```

Note: `anyio.open_process` on Windows with `shell=True` needs the command as a string. On Unix, pass `argv` as a list with no shell.

**python/src/gemini_sdk/process/process_manager.py:**
```python
import os
import signal
import sys
from typing import Any
import anyio
from anyio.abc import Process

from .binary_resolver import resolve_binary
from .env_builder import build_env
from .spawn_per_call import SpawnPerCallStrategy
from .process_strategy import ProcessStrategy


class ProcessManager:
    def __init__(self, strategy: ProcessStrategy | None = None) -> None:
        self._strategy = strategy or SpawnPerCallStrategy()

    async def spawn(
        self,
        *,
        argv: list[str] | None = None,
        cli_path: str | None = None,
        env: dict[str, str] | None = None,
        **kwargs: Any,
    ) -> Process:
        binary = resolve_binary(cli_path)
        clean_env = build_env(env)
        full_argv = [binary, *(argv or [])]
        return await self._strategy.spawn(full_argv, clean_env, **kwargs)


async def kill_tree(pid: int, grace_period_s: float = 5.0) -> None:
    """Kill a process and its entire tree. SIGTERM->grace->SIGKILL on Unix, taskkill /T /F on Windows."""
    if sys.platform == "win32":
        # Windows: taskkill /T kills tree, /F forces, no SIGTERM concept
        import subprocess as sp
        sp.run(["taskkill", "/T", "/F", "/PID", str(pid)], capture_output=True)
        return

    # Unix: SIGTERM -> grace -> SIGKILL
    try:
        os.kill(pid, signal.SIGTERM)
    except ProcessLookupError:
        return  # already dead

    # Wait grace period, then escalate to SIGKILL
    with anyio.move_on_after(grace_period_s):
        while True:
            try:
                os.kill(pid, 0)  # check alive
            except ProcessLookupError:
                return  # died during grace
            await anyio.sleep(0.1)

    # Still alive after grace — SIGKILL
    try:
        os.kill(pid, signal.SIGKILL)
    except ProcessLookupError:
        pass  # died between check and kill
```

try:

    # Kill children first, then parent
