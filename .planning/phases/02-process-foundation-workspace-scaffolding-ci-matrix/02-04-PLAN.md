**scripts/sync-version.sh:**
```bash
#!/usr/bin/env bash
# Syncs the root VERSION file into ts/package.json and python/pyproject.toml.
# Usage: bash scripts/sync-version.sh
# Run before publish or in CI release step.
set -euo pipefail

REPO_ROOT="$(cd "$(dirname "$0")/.." && pwd)"
VERSION_FILE="$REPO_ROOT/VERSION"

if [[ ! -f "$VERSION_FILE" ]]; then
  echo "ERROR: VERSION file not found at $VERSION_FILE" >&2
  exit 1
fi

VERSION=$(tr -d '[:space:]' < "$VERSION_FILE")

if [[ -z "$VERSION" ]]; then
  echo "ERROR: VERSION file is empty" >&2
  exit 1
fi

# Patch ts/package.json using node (handles JSON correctly)
node -e "
const fs = require('fs');
const path = '$REPO_ROOT/ts/package.json';
const pkg = JSON.parse(fs.readFileSync(path, 'utf8'));
pkg.version = '$VERSION';
fs.writeFileSync(path, JSON.stringify(pkg, null, 2) + '\n');
"

# Patch python/pyproject.toml using sed
# Match: version = "anything" and replace with version = "VERSION"
sed -i.bak "s/^version = \".*\"/version = \"${VERSION}\"/" "$REPO_ROOT/python/pyproject.toml"
rm -f "$REPO_ROOT/python/pyproject.toml.bak"

echo "Synced version ${VERSION} to ts/package.json and python/pyproject.toml"
```

IMPORTANT: The `sed -i.bak` + `rm .bak` pattern ensures portability across GNU sed and macOS BSD sed (BSD sed requires `-i ''` or `-i .bak`; `.bak` works on both).

**scripts/diff-test-names.sh:**
```bash
#!/usr/bin/env bash
# Enforces TS/Python test name parity (PAR-03).
# Extracts test descriptions from TS spec files and Python test docstrings.
# Exits nonzero if they diverge.
# Usage: bash scripts/diff-test-names.sh
set -euo pipefail

REPO_ROOT="$(cd "$(dirname "$0")/.." && pwd)"

TS_DIR="$REPO_ROOT/ts/src"
PY_DIR="$REPO_ROOT/python/tests"

TS_TESTS=$(mktemp)
PY_TESTS=$(mktemp)
trap "rm -f '$TS_TESTS' '$PY_TESTS'" EXIT

# Extract TS test names: test('description') or it('description')
# Handles both single and double quotes
grep -rh --include="*.spec.ts" --include="*.test.ts" \
  -oP "(?:test|it)\(['\"]([^'\"]+)['\"]" "$TS_DIR" 2>/dev/null \
  | sed "s/^\\(test\\|it\\)(['\"]//;s/['\"]$//" \
  | sort > "$TS_TESTS"

# Extract Python test docstrings (first line of each test function's docstring)
python3 - "$PY_DIR" <<'PYEOF' | sort > "$PY_TESTS"
import ast
import pathlib
import sys

test_dir = pathlib.Path(sys.argv[1])
for f in sorted(test_dir.rglob("test_*.py")):
    try:
        tree = ast.parse(f.read_text(encoding="utf-8"))
    except SyntaxError:
        continue
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            if node.name.startswith("test_"):
                doc = ast.get_docstring(node)
                if doc:
                    print(doc.split("\n")[0].strip())
                else:
                    # No docstring — emit function name as fallback (will likely fail parity)
                    print(f"MISSING_DOCSTRING: {node.name}")
PYEOF

TS_COUNT=$(wc -l < "$TS_TESTS")
PY_COUNT=$(wc -l < "$PY_TESTS")

echo "TS tests found: $TS_COUNT"
echo "Python tests found: $PY_COUNT"

if [[ "$TS_COUNT" -eq 0 ]] && [[ "$PY_COUNT" -eq 0 ]]; then
  echo "OK: No tests found in either language (expected during scaffolding)."
  exit 0
fi

if diff "$TS_TESTS" "$PY_TESTS"; then
  echo "OK: TS and Python test names match ($TS_COUNT tests)."
  exit 0
else
  echo ""
  echo "ERROR: TS and Python test names diverge. Fix parity before merging."
  echo "  TS tests:     $TS_TESTS (inspect with: cat $TS_TESTS)"
  echo "  Python tests: $PY_TESTS (inspect with: cat $PY_TESTS)"
  exit 1
fi
```

Create `.github/workflows/ci.yml` with the representative ~7-job test matrix plus a parity check job.

The matrix uses `include` (explicit list), NOT `exclude` from a cross-product. This matches the CONTEXT.md decision for a representative subset (~12 jobs total across test + parity).

```yaml
name: CI

on:
  push:
    branches: [main, dev]
  pull_request:
    branches: [main, dev]

jobs:
  test:
    name: "${{ matrix.os }} / Node ${{ matrix.node }} / Python ${{ matrix.python }}${{ matrix.locale && ' / ' || '' }}${{ matrix.locale || '' }}"
    runs-on: ${{ matrix.os }}
    strategy:
      fail-fast: false
      matrix:
        include:
          # Each OS: latest Node + latest Python
          - { os: ubuntu-latest,  node: "22", python: "3.13" }
          - { os: macos-latest,   node: "22", python: "3.13" }
          - { os: windows-latest, node: "22", python: "3.13" }
          # Oldest Node (18) on Linux
          - { os: ubuntu-latest,  node: "18", python: "3.13" }
          # Node 20 on Linux
          - { os: ubuntu-latest,  node: "20", python: "3.13" }
          # Oldest Python (3.10) on Linux
          - { os: ubuntu-latest,  node: "22", python: "3.10" }
          # Windows non-en-US: simulated ja-JP via chcp 932 (PLT-05)
          - { os: windows-latest, node: "22", python: "3.13", locale: "ja-JP" }

    steps:
      - uses: actions/checkout@v4

      - name: Install pnpm
        uses: pnpm/action-setup@v4
        with:
          version: 9

      - name: Setup Node.js ${{ matrix.node }}
        uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node }}
          cache: pnpm
          cache-dependency-path: ts/pnpm-lock.yaml

      - name: Setup Python ${{ matrix.python }}
        uses: actions/setup-python@v5
        with:
          python-version: ${{ matrix.python }}

      - name: Install uv
        uses: astral-sh/setup-uv@v5

      - name: Set Japanese code page (Windows ja-JP only)
        if: matrix.locale == 'ja-JP'
        shell: pwsh
        run: |
          # Force code page 932 (Shift_JIS) to exercise CJK mojibake vectors
          # Full locale change requires reboot (not CI-viable)
          # PYTHONUTF8=1 below ensures Python subprocess UTF-8 despite cp932
          chcp.com 932
          Write-Host "Active code page: $(chcp.com)"

      - name: Install TS dependencies
        run: cd ts && pnpm install --frozen-lockfile
        # Use --frozen-lockfile in CI to prevent lockfile changes

      - name: Run TS tests
        env:
          PYTHONUTF8: "1"
        run: cd ts && pnpm test

      - name: Install Python dependencies
        run: cd python && uv sync

      - name: Run Python tests
        env:
          PYTHONUTF8: "1"
          PYTHONIOENCODING: "utf-8"
        run: cd python && uv run pytest --tb=short

  parity:
    name: "Parity (TS <-> Python test names)"
    runs-on: ubuntu-latest
    needs: []  # Runs independently of test matrix
    steps:
      - uses: actions/checkout@v4

      - name: Setup Python (for AST parsing)
        uses: actions/setup-python@v5
        with:
          python-version: "3.13"

      - name: Check test name parity
        run: bash scripts/diff-test-names.sh

  version-sync:
    name: "Version sync check"
    runs-on: ubuntu-latest
    needs: []
    steps:
      - uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: "22"

      - name: Run sync-version and check for drift
        run: |
          bash scripts/sync-version.sh
          # If sync-version.sh changed any files, there's version drift
          if ! git diff --quiet ts/package.json python/pyproject.toml; then
            echo "ERROR: Version drift detected. Run 'bash scripts/sync-version.sh' and commit."
            git diff ts/package.json python/pyproject.toml
            exit 1
          fi
          echo "OK: Versions in sync."
```

Key design decisions per CONTEXT.md:
- `fail-fast: false` — let all jobs complete even if one fails
- Windows is NOT `continue-on-error` — it must pass (PLT-03)
- `pnpm/action-setup@v4` with version 9 — matches pnpm 9.x from RESEARCH.md
- `astral-sh/setup-uv@v5` — official uv GitHub Action
- `--frozen-lockfile` for pnpm in CI to prevent unexpected lockfile changes
- `needs: []` on parity job — runs independently, blocks merge via branch protection rules (user configures in GitHub settings)
- Both `PYTHONUTF8=1` and `PYTHONIOENCODING=utf-8` set in env for all test steps (FDN-04)
- `cache-dependency-path` points to `ts/pnpm-lock.yaml` for proper pnpm cache

Note: The `--frozen-lockfile` flag requires `pnpm-lock.yaml` to exist. On first run, change to `pnpm install` without the flag, or ensure the lockfile is committed. The executor should check if `ts/pnpm-lock.yaml` exists and adjust accordingly.
