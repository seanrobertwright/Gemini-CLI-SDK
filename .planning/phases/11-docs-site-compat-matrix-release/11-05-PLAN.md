  **1. Create `.github/workflows/docs.yml`** (EXACT content):

  ```yaml
  name: Deploy Docs

  on:
    push:
      branches: [master]
      paths:
        - 'docs/**'
        - 'docs-python/**'
        - 'docs/known-issues.yml'
        - 'ts/src/**'
        - 'python/src/**'
        - 'typedoc.json'
        - 'mkdocs.yml'
        - 'docs/.vitepress/**'
        - '.github/workflows/docs.yml'
    workflow_dispatch:

  permissions:
    contents: read
    pages: write
    id-token: write

  concurrency:
    group: pages
    cancel-in-progress: false

  jobs:
    build:
      runs-on: ubuntu-latest
      steps:
        - uses: actions/checkout@v4
        - name: Setup pnpm
          uses: pnpm/action-setup@v4
          with:
            version: 9
        - name: Setup Node
          uses: actions/setup-node@v4
          with:
            node-version: '22'
            cache: 'pnpm'
        - name: Setup Python
          uses: actions/setup-python@v5
          with:
            python-version: '3.13'
        - name: Setup uv
          uses: astral-sh/setup-uv@v5
        - name: Configure Pages
          uses: actions/configure-pages@v5
        - name: Install Python docs deps
          run: |
            cd python
            uv sync --extra docs
        - name: Build Python docs (mkdocs)
          run: uv run --project python mkdocs build --strict --config-file mkdocs.yml
        - name: Install TS deps
          run: pnpm install --frozen-lockfile
        - name: Generate TypeDoc markdown
          run: pnpm docs:typedoc
        - name: Build VitePress
          env:
            NODE_OPTIONS: --max_old_space_size=4096
          run: pnpm docs:build
        - name: Upload Pages artifact
          uses: actions/upload-pages-artifact@v3
          with:
            path: docs/.vitepress/dist

    deploy:
      needs: build
      runs-on: ubuntu-latest
      environment:
        name: github-pages
        url: ${{ steps.deployment.outputs.page_url }}
      steps:
        - name: Deploy to GitHub Pages
          id: deployment
          uses: actions/deploy-pages@v4
  ```

  **2. Create `.github/workflows/release.yml`** (EXACT content — changesets npm publish):

  ```yaml
  name: Release (npm)

  on:
    push:
      branches: [master]

  concurrency:
    group: release-${{ github.ref }}
    cancel-in-progress: false

  jobs:
    release:
      runs-on: ubuntu-latest
      permissions:
        contents: write      # create version-bump PR + push tags
        pull-requests: write # open/update the "Version Packages" PR
        id-token: write      # reserved for future npm OIDC
      steps:
        - uses: actions/checkout@v4
          with:
            fetch-depth: 0
        - name: Setup pnpm
          uses: pnpm/action-setup@v4
          with:
            version: 9
        - name: Setup Node
          uses: actions/setup-node@v4
          with:
            node-version: '22'
            cache: 'pnpm'
            registry-url: 'https://registry.npmjs.org'
        - name: Install deps
          run: pnpm install --frozen-lockfile
        - name: Run tests before release
          run: pnpm -C ts test --run
        - name: Create Release Pull Request or Publish
          id: changesets
          uses: changesets/action@v1
          with:
            version: pnpm run version-packages
            publish: pnpm run ci:publish
            commit: "chore(release): version packages"
            title: "chore(release): version packages"
          env:
            GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
            NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}
        - name: Mirror CHANGELOG to Python
          if: steps.changesets.outputs.published == 'true'
          run: |
            bash scripts/mirror-changelog.sh
            if ! git diff --quiet python/CHANGELOG.md; then
              git config user.name "github-actions[bot]"
              git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
              git add python/CHANGELOG.md
              git commit -m "chore(release): mirror CHANGELOG to python"
              git push
            fi
  ```

  **3. Create `.github/workflows/pypi-publish.yml`** (EXACT content — PyPI trusted publishing):

  ```yaml
  name: Publish to PyPI

  on:
    release:
      types: [created]
    workflow_dispatch:
      inputs:
        dry_run:
          description: "Skip actual publish (build only)"
          required: false
          default: "false"

  jobs:
    publish:
      runs-on: ubuntu-latest
      environment:
        name: release
        url: https://pypi.org/project/gemini-sdk/
      permissions:
        id-token: write  # required for PyPI trusted publishing OIDC
        contents: read
      steps:
        - uses: actions/checkout@v4
        - name: Setup Python
          uses: actions/setup-python@v5
          with:
            python-version: '3.13'
        - name: Setup uv
          uses: astral-sh/setup-uv@v5
        - name: Verify PyPI name available / owned
          run: bash scripts/check-pypi-name.sh
        - name: Sync version file into pyproject
          run: bash scripts/sync-version.sh
        - name: Build distribution
          run: |
            cd python
            uv build
        - name: List built artifacts
          run: ls -la python/dist/
        - name: Publish to PyPI (trusted publishing)
          if: github.event_name == 'release' || inputs.dry_run != 'true'
          run: |
            cd python
            uv publish --trusted-publishing always
        - name: Dry-run notice
          if: github.event_name == 'workflow_dispatch' && inputs.dry_run == 'true'
          run: echo "DRY RUN — skipped uv publish. Artifacts in python/dist/ ready for manual upload if needed."
  ```
