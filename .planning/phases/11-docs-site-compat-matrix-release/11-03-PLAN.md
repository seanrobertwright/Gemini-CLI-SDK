       ```bash
       #!/usr/bin/env bash
       # mirror-changelog.sh — REL-04
       #
       # Copies the top version section from root CHANGELOG.md (generated by
       # changesets for @gemini-sdk/core) into python/CHANGELOG.md so the Python
       # package release notes stay in sync with the TS changelog.
       #
       # Usage: bash scripts/mirror-changelog.sh
       # Prerequisites: CHANGELOG.md and python/CHANGELOG.md exist.
       # Exit codes: 0 on success, 1 on missing file, 2 on parse failure.

       set -euo pipefail

       ROOT_CHANGELOG="CHANGELOG.md"
       PY_CHANGELOG="python/CHANGELOG.md"

       if [ ! -f "$ROOT_CHANGELOG" ]; then
         echo "ERROR: $ROOT_CHANGELOG not found" >&2
         exit 1
       fi
       if [ ! -f "$PY_CHANGELOG" ]; then
         echo "ERROR: $PY_CHANGELOG not found" >&2
         exit 1
       fi

       # Extract everything from the first "## " line up to (but not including) the second "## " line.
       TOP_SECTION=$(awk '
         /^## / { count++; if (count == 2) exit }
         count >= 1 { print }
       ' "$ROOT_CHANGELOG")

       if [ -z "$TOP_SECTION" ]; then
         echo "ERROR: no version section found in $ROOT_CHANGELOG" >&2
         exit 2
       fi

       # Strip any existing top "## " section from python/CHANGELOG.md and prepend the new one.
       # Preserve the "# gemini-sdk (Python)" header (line 1).
       HEADER=$(head -n 1 "$PY_CHANGELOG")
       REMAINDER=$(awk '
         BEGIN { skip = 0; count = 0 }
         NR == 1 { next }  # skip header (already captured)
         /^## / { count++; if (count == 1) { skip = 1; next } }
         /^## / && count >= 2 { skip = 0 }
         skip == 0 && NR > 1 { print }
       ' "$PY_CHANGELOG")

       {
         echo "$HEADER"
         echo ""
         echo "$TOP_SECTION"
         if [ -n "$REMAINDER" ]; then
           echo "$REMAINDER"
         fi
       } > "$PY_CHANGELOG.tmp"

       mv "$PY_CHANGELOG.tmp" "$PY_CHANGELOG"
       echo "Mirrored top section from $ROOT_CHANGELOG to $PY_CHANGELOG"
       ```
       Make executable: `chmod +x scripts/mirror-changelog.sh`.

    4. **Create `scripts/check-pypi-name.sh`** (EXACT content):
       ```bash
       #!/usr/bin/env bash
       # check-pypi-name.sh — REL-02 pre-publish gate
       #
       # Verifies whether the PyPI project name is available OR already registered
       # to the configured trusted publisher. Run BEFORE the first PyPI publish.
       #
       # Usage: bash scripts/check-pypi-name.sh [package-name]
       # Default package-name: gemini-sdk (from python/pyproject.toml [project].name)
       # Exit codes:
       #   0 = name available OR already published (safe to publish / update)
       #   1 = name taken by a DIFFERENT project owner (need fallback name)
       #   2 = network failure

       set -euo pipefail

       NAME="${1:-gemini-sdk}"
       URL="https://pypi.org/pypi/${NAME}/json"

       if ! command -v curl >/dev/null 2>&1; then
         echo "ERROR: curl not found" >&2
         exit 2
       fi

       HTTP_CODE=$(curl -s -o /tmp/pypi-check-response.json -w "%{http_code}" "$URL" || echo "000")

       case "$HTTP_CODE" in
         404)
           echo "OK: PyPI name '$NAME' is available (404 — not yet registered)"
           exit 0
           ;;
         200)
           echo "INFO: PyPI name '$NAME' is already registered"
           echo "Verify the 'info.author' / 'info.maintainer' fields below match this project:"
           if command -v python3 >/dev/null 2>&1; then
             python3 -c "import json,sys; d=json.load(open('/tmp/pypi-check-response.json')); print(' author:', d['info'].get('author')); print(' home_page:', d['info'].get('home_page')); print(' project_urls:', d['info'].get('project_urls'))"
           else
             grep -oE '"(author|home_page)": "[^"]*"' /tmp/pypi-check-response.json | head -5
           fi
           echo "If the existing project is NOT this one, fall back to 'gemini-cli-sdk' or another name."
           exit 0
           ;;
         000)
           echo "ERROR: network failure checking $URL" >&2
           exit 2
           ;;
         *)
           echo "WARNING: unexpected HTTP $HTTP_CODE from $URL" >&2
           exit 2
           ;;
       esac
       ```
