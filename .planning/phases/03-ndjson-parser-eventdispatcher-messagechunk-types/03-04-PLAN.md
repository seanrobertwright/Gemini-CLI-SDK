**parse_ndjson.py** — Mechanical port of TS parseNdjson:

```python
from __future__ import annotations
import json
from typing import AsyncIterator, Union
from .types import RawEvent, KNOWN_RAW_TYPES

MAX_LINE = 1024 * 1024  # 1 MiB (PRS-01)

async def parse_ndjson(
    stream: Union["anyio.abc.ByteReceiveStream", "AsyncIterator[bytes]"],
) -> AsyncIterator[RawEvent]:
    """Stage 1: byte stream -> RawEvent async iterator."""
    buf = b""
    async for chunk in stream:
        buf += chunk
        while b"\n" in buf:
            line_bytes, buf = buf.split(b"\n", 1)
            line = line_bytes.rstrip(b"\r").decode("utf-8", errors="replace")
            if not line:
                continue
            yield _parse_line(line)
        if len(buf) > MAX_LINE:
            yield {"type": "cli_log", "line": buf[:MAX_LINE].decode("utf-8", errors="replace")}
            buf = buf[MAX_LINE:]
    # Flush remaining buffer
    if buf:
        line = buf.rstrip(b"\r").decode("utf-8", errors="replace")
        if line:
            yield _parse_line(line)

def _parse_line(line: str) -> RawEvent:
    try:
        obj = json.loads(line)
    except (json.JSONDecodeError, ValueError):
        return {"type": "cli_log", "line": line}
    if not isinstance(obj, dict) or "type" not in obj:
        return {"type": "cli_log", "line": line}
    if obj["type"] not in KNOWN_RAW_TYPES:
        return {"type": "unknown", "raw": obj}
    return obj
```

**dispatch.py** — Mechanical port of TS dispatch:

```python
from __future__ import annotations
from typing import AsyncIterator
from .types import RawEvent, MessageChunk

async def dispatch(events: AsyncIterator[RawEvent]) -> AsyncIterator[MessageChunk]:
    """Stage 2: RawEvent -> MessageChunk async iterator with tool pairing."""
    pending: dict[str, dict] = {}  # tool_id -> ToolChunk
    session_id = ""

    async for event in events:
        event_type = event.get("type", "")

        if event_type == "init":
            session_id = event["session_id"]
            yield {"type": "system", "subtype": "init", "sessionId": event["session_id"], "model": event["model"]}

        elif event_type == "message":
            if _is_thinking(event):
                yield {"type": "thinking", "content": event["content"]}
            elif event.get("role") == "assistant":
                yield {"type": "assistant", "content": event["content"]}
            else:
                yield {"type": "system", "subtype": "message", "role": event["role"], "content": event["content"]}

        elif event_type == "tool_use":
            pending[event["tool_id"]] = {
                "type": "tool",
                "toolName": event["tool_name"],
                "toolId": event["tool_id"],
                "parameters": event["parameters"],
            }

        elif event_type == "tool_result":
            tool_chunk = pending.pop(event["tool_id"], None)
            if tool_chunk is not None:
                yield tool_chunk
            result_chunk: dict = {
                "type": "tool_result",
                "toolId": event["tool_id"],
                "status": event["status"],
                "output": event["output"],
            }
            if "error" in event and event["error"] is not None:
                result_chunk["error"] = event["error"]
            yield result_chunk

        elif event_type == "error":
            if _is_rate_limit_error(event):
                err = event["error"]
                yield {
                    "type": "rate_limit",
                    "code": err.get("code", 429),
                    "message": err.get("message", ""),
                    "status": err.get("status"),
                }
            else:
                raise RuntimeError(f"Unhandled error event: {event.get('error')}")

        elif event_type == "result":
            yield {
                "type": "result",
                "sessionId": session_id,
                "stopReason": _map_stop_reason(event["status"]),
            }
        # unknown and cli_log: silently skip

    # Flush unpaired tool_use on stream end (PRS-07)
    for tool_chunk in pending.values():
        yield {**tool_chunk, "incomplete": True}

def _is_thinking(event: dict) -> bool:
    return event.get("thought") is True or event.get("role") == "thinking"

def _is_rate_limit_error(event: dict) -> bool:
    err = event.get("error", {})
    return err.get("code") == 429 or err.get("status") == "RESOURCE_EXHAUSTED"

def _map_stop_reason(status: str) -> str:
    return "end_turn" if status == "success" else status
```
