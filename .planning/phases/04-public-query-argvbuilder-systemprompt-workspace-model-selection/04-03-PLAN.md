export function buildArgv(options: QueryOptions): string[] {
  const argv: string[] = ['--output-format', 'stream-json', '-p', options.prompt];
  if (options.model !== undefined && options.model !== 'auto') {
    argv.push('--model', options.model as string);
  }
  if (options.additionalDirectories?.length) {
    for (const dir of options.additionalDirectories) {
      argv.push('--include-directories', dir);
    }
  }
  return argv;
}

**Step 3: Create python/src/gemini_sdk/query/build_argv.py**

```python
from __future__ import annotations
from typing import Any
from .types import QueryOptions

def build_argv(options: QueryOptions) -> list[str]:
    argv: list[str] = [
        "--output-format", "stream-json",
        "-p", options["prompt"],
    ]
    model = options.get("model")
    if model is not None and str(model) != "auto":
        argv.extend(["--model", str(model)])
    dirs = options.get("additional_directories")
    if dirs:
        for d in dirs:
            argv.extend(["--include-directories", d])
    return argv
```

**Step 4: Create python/src/gemini_sdk/query/query.py**

Implement three functions:
- `async def query(options: QueryOptions) -> AsyncIterator[MessageChunk]` — async generator
- `async def query_raw(options: QueryOptions) -> AsyncIterator[RawEvent]` — async generator, no dispatch
- `async def query_full(options: QueryOptions) -> QueryResult` — accumulator

Key implementation details:
- `_write_temp_system_prompt(system_prompt)`: uses `secrets.token_hex(8)` for suffix, `os.path.join(tempfile.gettempdir(), f'gemini-sdk-system-{suffix}.md')`, writes with `anyio.Path(path).write_text(content, encoding='utf-8')`
- `query()` lifecycle:
  1. Check `cancel_scope.cancel_called` if present — raise AbortError if already cancelled
  2. Write temp system prompt → get temp_path
  3. Build env_overrides with GEMINI_SYSTEM_MD if temp_path
  4. Call `build_argv(options)` → argv
  5. `manager = ProcessManager(); proc = await manager.spawn(argv=argv, cli_path=options.get('cli_path'), env={**(options.get('env') or {}), **env_overrides}, cwd=options.get('cwd'))`
  6. Try: `async for chunk in dispatch(parse_ndjson(proc.stdout)):` — check cancel, capture model from init, enrich result chunk, yield
  7. Finally: `kill_tree(proc.pid)` try/except pass, `anyio.Path(temp_path).unlink(missing_ok=True)` try/except pass
- Model downgrade: only compare when `requested_model` is not None and not 'auto'; add `requestedModel`/`actualModel` to result dict when mismatch
- Cancel: check `cancel_scope.cancel_called` after each yield; if cancelled raise AbortError after cleanup
- `query_full()`: iterate `query()`, concat assistant content, capture result fields, return QueryResult dict
