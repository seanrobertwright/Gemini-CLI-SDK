    === Step A: Write `scripts/gen-errors.mjs` (Node ESM, ~100 LOC) ===

    Structure:
    ```js
    #!/usr/bin/env node
    // scripts/gen-errors.mjs
    // Reads spec/errors.yaml, writes ts/src/errors/errors.ts.
    // Naming: YAML retry_after_ms_source → TS retryAfterMs.
    import { readFileSync, writeFileSync } from 'node:fs';
    import { fileURLToPath } from 'node:url';
    import { dirname, resolve } from 'node:path';
    import yaml from 'js-yaml';

    const __dirname = dirname(fileURLToPath(import.meta.url));
    const REPO_ROOT = resolve(__dirname, '..');
    const YAML_PATH = resolve(REPO_ROOT, 'spec/errors.yaml');
    const OUT_PATH = resolve(REPO_ROOT, 'ts/src/errors/errors.ts');

    const doc = yaml.load(readFileSync(YAML_PATH, 'utf8'));

    const HEADER = `// AUTO-GENERATED by scripts/gen-errors.mjs — DO NOT EDIT DIRECTLY\n// Source: spec/errors.yaml (version ${doc.version})\n// Run: node scripts/gen-errors.mjs\n\n`;

    let out = HEADER;
    out += `export type ArchonBucket = ${doc.buckets.map(b => `'${b}'`).join(' | ')};\n\n`;

    for (const entry of doc.errors) {
      const baseClass = entry.base === 'Error' ? 'Error' : entry.base;
      const defaultMsg = entry.message_template ? JSON.stringify(entry.message_template) : 'undefined';
      const retryableLiteral = entry.retryable ? 'true' : 'false';
      const bucket = `'${entry.bucket}'`;

      out += `export class ${entry.name} extends ${baseClass} {\n`;
      out += `  readonly bucket: ArchonBucket = ${bucket};\n`;
      out += `  readonly retryable: boolean = ${retryableLiteral};\n`;
      out += `  retryAfterMs?: number;\n\n`;
      out += `  constructor(message?: string, options?: { retryAfterMs?: number }) {\n`;
      out += `    super(message ?? ${defaultMsg});\n`;
      out += `    this.name = '${entry.name}';\n`;
      out += `    Object.setPrototypeOf(this, new.target.prototype);\n`;
      out += `    if (options?.retryAfterMs !== undefined) this.retryAfterMs = options.retryAfterMs;\n`;
      out += `  }\n`;
      out += `}\n\n`;
    }

    writeFileSync(OUT_PATH, out, 'utf8');
    console.log(`Wrote ${OUT_PATH} (${doc.errors.length} classes)`);
    ```

    Add to `ts/package.json` scripts: `"gen:errors": "node ../scripts/gen-errors.mjs"`.

    === Step B: Write `scripts/gen-errors.py` (~80 LOC) ===

    Structure (equivalent semantics; Python naming snake_case):
    ```python
    #!/usr/bin/env python3
    # scripts/gen-errors.py
    # Reads spec/errors.yaml, writes python/src/gemini_sdk/errors/errors.py.
    # Naming: YAML retry_after_ms_source → Python retry_after_ms.
    from __future__ import annotations
    import pathlib, yaml

    REPO_ROOT = pathlib.Path(__file__).resolve().parent.parent
    YAML_PATH = REPO_ROOT / "spec" / "errors.yaml"
    OUT_PATH = REPO_ROOT / "python" / "src" / "gemini_sdk" / "errors" / "errors.py"

    doc = yaml.safe_load(YAML_PATH.read_text(encoding="utf-8"))

    header = (
        f"# AUTO-GENERATED by scripts/gen-errors.py — DO NOT EDIT DIRECTLY\n"
        f"# Source: spec/errors.yaml (version {doc['version']})\n"
        f"# Run: python scripts/gen-errors.py\n\n"
        "from __future__ import annotations\n"
        "from typing import Literal, Optional\n\n"
    )
    out = header
    buckets = ", ".join(f"'{b}'" for b in doc["buckets"])
    out += f"ArchonBucket = Literal[{buckets}]\n\n"

    for entry in doc["errors"]:
        base = "Exception" if entry["base"] == "Error" else entry["base"]
        default_msg = repr(entry.get("message_template", ""))
        retryable = "True" if entry.get("retryable") else "False"
        bucket = repr(entry["bucket"])
        out += f"class {entry['name']}({base}):\n"
        out += f"    bucket: ArchonBucket = {bucket}\n"
        out += f"    retryable: bool = {retryable}\n"
        out += f"    retry_after_ms: Optional[int] = None\n\n"
        out += f"    def __init__(self, message: str = {default_msg}, *, retry_after_ms: Optional[int] = None) -> None:\n"
        out += f"        super().__init__(message)\n"
        out += f"        if retry_after_ms is not None:\n"
        out += f"            self.retry_after_ms = retry_after_ms\n\n"

    OUT_PATH.write_text(out, encoding="utf-8")
    print(f"Wrote {OUT_PATH} ({len(doc['errors'])} classes)")
    ```
