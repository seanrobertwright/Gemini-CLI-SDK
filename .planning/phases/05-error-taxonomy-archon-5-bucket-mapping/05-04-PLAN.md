    ```bash
    #!/usr/bin/env bash
    # scripts/lint-errors.sh
    # ERR-07 + PAR-05 drift detection for the error taxonomy.
    # Exits 0 when YAML ↔ TS ↔ Python are in sync; non-zero otherwise.
    #
    # Compat: macOS BSD grep (no -P). Git Bash on Windows. Use grep -E only.
    #
    set -euo pipefail

    REPO_ROOT="$(cd "$(dirname "$0")/.." && pwd)"
    cd "$REPO_ROOT"

    YAML_PATH="spec/errors.yaml"
    TS_GEN="ts/src/errors/errors.ts"
    PY_GEN="python/src/gemini_sdk/errors/errors.py"

    # --- 1. Re-run codegen ---
    echo "[lint-errors] Regenerating TS..."
    node scripts/gen-errors.mjs >/dev/null
    echo "[lint-errors] Regenerating Python..."
    if command -v uv >/dev/null 2>&1; then
      (cd python && uv run python ../scripts/gen-errors.py >/dev/null)
    else
      python3 scripts/gen-errors.py >/dev/null
    fi

    # --- 2. Drift check ---
    if ! git diff --exit-code -- "$TS_GEN" "$PY_GEN"; then
      echo "[lint-errors] FAIL: generated files drifted from committed versions." >&2
      echo "  Run: node scripts/gen-errors.mjs && (cd python && uv run python ../scripts/gen-errors.py)" >&2
      echo "  Then commit the updated files." >&2
      exit 1
    fi

    # --- 3. Extract class names ---
    # YAML names: `  - name: Foo` entries under `errors:`
    yaml_names="$(grep -E '^\s*-\s*name:\s*[A-Za-z0-9_]+' "$YAML_PATH" | grep -oE '[A-Za-z0-9_]+$' | sort -u)"

    # TS names: `export class Foo extends ...`
    ts_names="$(grep -oE '^export class [A-Za-z0-9_]+' "$TS_GEN" | awk '{print $3}' | sort -u)"

    # Python names: `class Foo(...):`
    py_names="$(grep -oE '^class [A-Za-z0-9_]+' "$PY_GEN" | awk '{print $2}' | sort -u)"

    # --- 4. Cross-check set equality ---
    fail=0
    diff_yaml_ts="$(comm -3 <(echo "$yaml_names") <(echo "$ts_names") || true)"
    if [ -n "$diff_yaml_ts" ]; then
      echo "[lint-errors] FAIL: YAML ↔ TS class set mismatch:" >&2
      echo "$diff_yaml_ts" >&2
      fail=1
    fi
    diff_yaml_py="$(comm -3 <(echo "$yaml_names") <(echo "$py_names") || true)"
    if [ -n "$diff_yaml_py" ]; then
      echo "[lint-errors] FAIL: YAML ↔ Python class set mismatch:" >&2
      echo "$diff_yaml_py" >&2
      fail=1
    fi

    if [ "$fail" -ne 0 ]; then
      exit 1
    fi

    yaml_count="$(echo "$yaml_names" | wc -l | tr -d ' ')"
    echo "[lint-errors] OK: $yaml_count classes in sync across YAML, TS, Python."
    ```

    ```ts
    import { describe, it, expect } from 'vitest';
    import { readFileSync, readdirSync } from 'node:fs';
    import { join, resolve } from 'node:path';
    import { ErrorMapper, type StreamErrorEvent } from '../src/errors/ErrorMapper.js';
    import * as errorClasses from '../src/errors/errors.js';

    const FIXTURES = resolve(__dirname, '../../spec/fixtures');
    const slugs = readdirSync(FIXTURES)
      .filter((f) => f.startsWith('error-') && f.endsWith('.ndjson'))
      .map((f) => f.replace('.ndjson', ''));

    for (const slug of slugs) {
      describe(`error corpus: ${slug}`, () => {
        const ndjsonPath = join(FIXTURES, `${slug}.ndjson`);
        const stderrPath = join(FIXTURES, `${slug}.stderr.txt`);
        const expectedPath = join(FIXTURES, `${slug}.expected.json`);

        const ndjson = readFileSync(ndjsonPath, 'utf8');
        const expected = JSON.parse(readFileSync(expectedPath, 'utf8'));
        const stderr = (() => {
          try { return readFileSync(stderrPath, 'utf8'); } catch { return ''; }
        })();

        const expectedClassName = expected._errorType as string | undefined;
        if (!expectedClassName) return;  // fixture doesn't test error classification

        const expectedCtor = (errorClasses as any)[expectedClassName];

        it(`run_corpus_${slug.replace(/-/g, '_')}_fromStreamEvent_matches_expected_type`, () => {
          // Extract the first {type:'error'} event from the ndjson, if any
          const lines = ndjson.split('\n').filter(Boolean);
          const errorEvt = lines
            .map((l) => { try { return JSON.parse(l); } catch { return null; } })
            .find((e) => e && e.type === 'error') as StreamErrorEvent | undefined;
          if (errorEvt) {
            const err = ErrorMapper.fromStreamEvent(errorEvt);
            expect(err).toBeInstanceOf(expectedCtor);
            expect(err.bucket).toBe(new expectedCtor().bucket);
            expect(err.retryable).toBe(new expectedCtor().retryable);
          } else {
            // No stream event → this slug tests only the exit path
            expect(true).toBe(true);
          }
        });

        it(`run_corpus_${slug.replace(/-/g, '_')}_fromExit_matches_expected_type`, () => {
          if (!stderr) {
            // abort-midstream has empty stderr; ERR-06 path asserted separately
            return;
          }
          const err = ErrorMapper.fromExit(1, stderr, []);
          expect(err).toBeInstanceOf(expectedCtor);
        });

        it(`run_corpus_${slug.replace(/-/g, '_')}_both_paths_agree_on_class`, () => {
          const lines = ndjson.split('\n').filter(Boolean);
          const errorEvt = lines
            .map((l) => { try { return JSON.parse(l); } catch { return null; } })
            .find((e) => e && e.type === 'error') as StreamErrorEvent | undefined;
          if (errorEvt && stderr) {
            const fromStream = ErrorMapper.fromStreamEvent(errorEvt);
            const fromExit = ErrorMapper.fromExit(1, stderr, []);
            expect(fromStream.constructor.name).toBe(fromExit.constructor.name);
          }
        });
      });
    }
    ```

    === Step B: `python/tests/errors/test_error_mapper_corpus.py` ===

    Pytest parametrize over the same glob.

    ```python
    from __future__ import annotations
    import json, pathlib
    import pytest
    from gemini_sdk.errors.error_mapper import ErrorMapper
    from gemini_sdk.errors import errors as error_classes

    FIXTURES = pathlib.Path(__file__).resolve().parents[2] / "spec" / "fixtures"

    def _slugs() -> list[str]:
        return sorted(
            p.stem for p in FIXTURES.glob("error-*.ndjson")
        )

    @pytest.mark.parametrize("slug", _slugs())
    def test_run_corpus_from_stream_event_matches_expected_type(slug: str):
        expected_path = FIXTURES / f"{slug}.expected.json"
        expected = json.loads(expected_path.read_text(encoding="utf-8"))
        expected_class_name = expected.get("_errorType")
        if not expected_class_name:
            pytest.skip(f"{slug}: no _errorType in expected.json")
        expected_cls = getattr(error_classes, expected_class_name)
        ndjson = (FIXTURES / f"{slug}.ndjson").read_text(encoding="utf-8")
        for line in ndjson.splitlines():
            if not line.strip():
                continue
            try:
                evt = json.loads(line)
            except Exception:
                continue
            if evt.get("type") == "error":
                err = ErrorMapper.from_stream_event(evt)
                assert isinstance(err, expected_cls), f"{slug}: got {type(err).__name__}, expected {expected_class_name}"
                return
        pytest.skip(f"{slug}: no stream error event to classify")

    @pytest.mark.parametrize("slug", _slugs())
    def test_run_corpus_from_exit_matches_expected_type(slug: str):
        expected_path = FIXTURES / f"{slug}.expected.json"
        expected = json.loads(expected_path.read_text(encoding="utf-8"))
        expected_class_name = expected.get("_errorType")
        if not expected_class_name:
            pytest.skip(f"{slug}: no _errorType in expected.json")
        stderr_path = FIXTURES / f"{slug}.stderr.txt"
        if not stderr_path.exists() or stderr_path.stat().st_size == 0:
            pytest.skip(f"{slug}: no stderr")
        stderr = stderr_path.read_text(encoding="utf-8")
        expected_cls = getattr(error_classes, expected_class_name)
        err = ErrorMapper.from_exit(1, stderr, [])
        assert isinstance(err, expected_cls), f"{slug}: got {type(err).__name__}, expected {expected_class_name}"

    @pytest.mark.parametrize("slug", _slugs())
    def test_run_corpus_both_paths_agree_on_class(slug: str):
        ndjson_path = FIXTURES / f"{slug}.ndjson"
        stderr_path = FIXTURES / f"{slug}.stderr.txt"
        if not stderr_path.exists():
            pytest.skip(f"{slug}: no stderr")
        stream_evt = None
        for line in ndjson_path.read_text(encoding="utf-8").splitlines():
            if not line.strip():
                continue
            try:
                evt = json.loads(line)
            except Exception:
                continue
            if evt.get("type") == "error":
                stream_evt = evt
                break
        if stream_evt is None:
            pytest.skip(f"{slug}: no stream error event")
        stderr = stderr_path.read_text(encoding="utf-8")
        from_stream = ErrorMapper.from_stream_event(stream_evt)
        from_exit = ErrorMapper.from_exit(1, stderr, [])
        assert type(from_stream).__name__ == type(from_exit).__name__, (
            f"{slug}: stream-path={type(from_stream).__name__}, exit-path={type(from_exit).__name__}"
        )
    ```
